



PROBLEM:- Insert an element in an array




#include <iostream>
using namespace std;

int main() {
    int A[100] = {10, 20, 30, 40, 50};
    int n = 5, pos, value;

    cout << "Enter position: ";
    cin >> pos;

    cout << "Enter value: ";
    cin >> value;

    for (int i = n; i >= pos; i--) {
        A[i] = A[i - 1];
    }

    A[pos - 1] = value;
    n++;

    cout << "Array after insertion: ";
    for (int i = 0; i < n; i++)
        cout << A[i] << " ";

    return 0;
}






PROBLEM:-  Delete an element from an array



#include <iostream>
using namespace std;

int main() {
    int A[100] = {10, 20, 30, 40, 50};
    int n = 5, pos;

    cout << "Enter position to delete: ";
    cin >> pos;

    for (int i = pos - 1; i < n - 1; i++) {
        A[i] = A[i + 1];
    }

    n--;

    cout << "Array after deletion: ";
    for (int i = 0; i < n; i++)
        cout << A[i] << " ";

    return 0;
}






PROBLEM:- Find the largest element


#include <iostream>
using namespace std;

int main() {
    int A[] = {10, 45, 23, 67, 12};
    int n = 5;

    int max = A[0];

    for (int i = 1; i < n; i++) {
        if (A[i] > max)
            max = A[i];
    }

    cout << "Largest element = " << max;

    return 0;
}
