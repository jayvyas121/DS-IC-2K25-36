PROBLEM:- Find the sum of all elements in a 2D array	


#include <iostream>
using namespace std;

int main() {
    int A[2][3] = {
        {1, 2, 3},
        {4, 5, 6}
    };

    int sum = 0;

    for (int i = 0; i < 2; i++) {
        for (int j = 0; j < 3; j++) {
            sum += A[i][j];
        }
    }

    cout << "Sum = " << sum;

    return 0;
}




PROBLEM:-Implement stack using an array



#include <iostream>
using namespace std;

#define MAX 5

class Stack {
    int A[MAX];
    int top;

public:
    Stack() {
        top = -1;
    }

    void push(int x) {
        if (top == MAX - 1) {
            cout << "Stack Overflow\n";
        }
        else {
            A[++top] = x;
        }
    }

    void pop() {
        if (top == -1) {
            cout << "Stack Underflow\n";
        }
        else {
            cout << "Deleted: " << A[top--] << endl;
        }
    }

    void peek() {
        if (top == -1)
            cout << "Stack is empty\n";
        else
            cout << "Top element: " << A[top] << endl;
    }

    void display() {
        for (int i = top; i >= 0; i--)
            cout << A[i] << " ";
        cout << endl;
    }
};

int main() {
    Stack s;

    s.push(10);
    s.push(20);
    s.push(30);
    s.push(40);
    s.push(50);

    s.display();

    s.peek();

    s.pop();

    s.display();

    return 0;
}







PROBLEM:- Create a stack with size M and put the element in this array A B C D E and find the peak element of the stack.



#include <iostream>
#include <stack>
using namespace std;

int main() {
    int M = 5;

    stack<char> s;

    char A[] = {'A', 'B', 'C', 'D', 'E'};

    // Push elements into stack
    for (int i = 0; i < M; i++) {
        s.push(A[i]);
    }

    // Peek/Top element
    cout << "Top element of stack: " << s.top() << endl;

    return 0;
}
